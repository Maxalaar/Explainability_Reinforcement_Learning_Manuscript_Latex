\begin{algorithm}[t]
\caption{Deep Critic-Only Reinforcement Learning}
\label{alg:critic_only}

\Input{
\\
\gls{mdp} $(\StateSet, \ObsSet, \ActionSet, \RewardSet, p)$\\
Architecture $\hat{q}$\\
Exploration rate $\varepsilon \in [0, 1]$\\
Discount factor $\gamma \in [0, 1]$\\
Learning rate $\eta > 0$\\
Initial parameters $\valueParams$ (optional)
}

\Output{
\\
Trained policy $\pi_{\valueParams}$
}

\AlgoRule

\uIf{$\valueParams$ is not given}{
    Initialize $\valueParams$ randomly\nllabel{alg:critic_only:init}\;
}

\BlankLine
\tcp{Exploration policy as defined in Equation~\eqref{eq:epsilon_greedy_policy}}
$\pi_{\valueParams} \leftarrow$ $\varepsilon$-greedy policy with respect to $\hat{q}_{\valueParams}$\nllabel{alg:critic_only:policy}

\BlankLine
\tcp{MSE loss function as defined in Equation~\eqref{eq:mse}}
$\mathcal{L} \leftarrow$ \gls{mse}\nllabel{alg:critic_only:loss}\;

\BlankLine
\While{stopping criterion not satisfied\nllabel{alg:critic_only:loop}}{
    \tcp{Data collection see Algorithm~\ref{alg:data_collection}}
    $\mathcal{D} \leftarrow$ \FnDataCollection{\gls{mdp}, $\pi_{\valueParams}, \gamma$}\nllabel{alg:critic_only:collect}

    \BlankLine
    \tcp{observation-action pairs and their returns, from $\mathcal{D}$}
    $\mathcal{D}' \leftarrow \big\{\big((o, a), g\big) \mid (s, o, a, g) \in \mathcal{D}\big\}$\nllabel{alg:critic_only:dataset}

    \BlankLine
    \tcp{Model update whith SGD see Algorithm~\ref{alg:stochastic_gradient_descent}}
    $\valueParams \leftarrow$ \FnSGD{$\mathcal{D}', \hat{q}, \mathcal{L}, \eta, \valueParams$}\nllabel{alg:critic_only:sgd}
}

\BlankLine
\tcp{Final policy without exploration as defined in Equation~\eqref{eq:greedy_policy}}
$\pi_{\valueParams} \leftarrow$ policy with respect to $\hat{q}_{\valueParams}$\nllabel{alg:critic_only:greedy}\;

\BlankLine
\Return{$\pi_{\valueParams}$}

\end{algorithm}
