% Mathematical notation, following Goodfellow et al. (deep learning) and
% Sutton & Barto (reinforcement learning). Change a style here only.

% Generic styles: vectors and matrices in bold upright, parameters in bold
% (theta: network or actor parameters, phi: value function / critic parameters).
\newcommand{\vect}[1]{\mathbf{#1}}
\newcommand{\mat}[1]{\mathbf{#1}}
\newcommand{\params}{\boldsymbol{\theta}}
\newcommand{\valueParams}{\boldsymbol{\phi}}

% Operators and metrics, set upright.
\DeclareMathOperator*{\argmax}{arg\,max}
\DeclareMathOperator*{\argmin}{arg\,min}
\DeclareMathOperator{\MSE}{MSE}
\DeclareMathOperator{\RI}{RI}
\DeclareMathOperator{\ARI}{ARI}
\newcommand{\BR}{\mathrm{BR}}
\newcommand{\CU}{\mathrm{CU}}
\newcommand{\AS}{\mathrm{AS}}
\newcommand{\ACU}{\mathrm{ACU}}

% Deep learning: per-example loss (#1 = parameters, #2 = example index).
\newcommand{\fullLoss}[2]{\mathcal{L}\left(\hat{f}_{#1}(\vect{x}^{(#2)}), \vect{y}^{(#2)}\right)}
\newcommand{\loss}[2]{\mathcal{L}^{(#2)}\left(#1\right)}

% Function computed by layer #1 of a network.
\newcommand{\LayerFunction}[1]{f^{(#1)}}

% Attention mechanism: softmax operator and dimension of queries and keys.
\DeclareMathOperator{\softmax}{softmax}
\newcommand{\KeyDim}{d_{\mathrm{key}}}

% Latent representation h^{(l)}: optional #1 = layer, optional #2 = decoration
% (e.g. \prime or *), placed in the same superscript.
\NewDocumentCommand{\latentRepresentation}{o o}{%
  \vect{h}\IfValueT{#1}{^{(#1)\IfValueT{#2}{#2}}}%
}

% Encoder, decoder and clustering functions (optional #1 = model index).
\NewDocumentCommand{\Encoder}{o}{f_{\mathrm{enc}\IfValueT{#1}{,#1}}}
\NewDocumentCommand{\Decoder}{o}{f_{\mathrm{dec}\IfValueT{#1}{,#1}}}
\newcommand{\ClusteringFunction}{f_{\mathrm{clust}}}

% Probe for concept #2 at layer #1.
\newcommand{\Probe}[2]{\psi^{(#1,#2)}_{\params}}

% Reinforcement learning: spaces of the MDP.
\newcommand{\StateSet}{\mathcal{S}}
\newcommand{\ObsSet}{\mathcal{O}}
\newcommand{\ActionSet}{\mathcal{A}}
\newcommand{\RewardSet}{\mathcal{R}}
\newcommand{\Distributions}[1]{\mathcal{P}(#1)}
\newcommand{\StateSetInitial}{\mathcal{S}^{\mathrm{init}}}
\newcommand{\StateSetTerminal}{\mathcal{S}^{\mathrm{ter}}}
\newcommand{\StateInitial}{s^{\mathrm{init}}}
\newcommand{\StateTerminal}{s^{\mathrm{ter}}}

% Reinforcement learning: optimal policy.
\newcommand{\OptimalPolicy}{\pi^{*}}

% Reinforcement learning: advantage function and its estimate at a time step.
\newcommand{\Adv}{\mathrm{Adv}}
\newcommand{\AdvEst}{\widehat{\mathrm{Adv}}}

% Distribution over states given an observation (#1 = policy).
\newcommand{\Belief}[1]{\beta_{#1}}

% Method: initial and surrogate policies, their encoders and latent sets.
\newcommand{\InitialPolicy}{\textcolor{mygreen}{\pi}}
\newcommand{\SurrogatePolicy}[1]{\textcolor{myred}{\pi'_{#1}}}
\NewDocumentCommand{\SurrogateEncoder}{o}{\textcolor{myblue}{\IfValueTF{#1}{\Encoder[#1]}{\Encoder}}}
\newcommand{\LatentSet}[1]{\textcolor{myblue}{\mathcal{H}_{#1}}}
\newcommand{\ObsData}{\mathcal{O}_{\mathrm{data}}}
\newcommand{\logits}{\boldsymbol{\ell}}

% Metrics: baseline clusterings and similarities (#1 = obs, act or untr).
% Optional #1 = index of an untrained network.
\NewDocumentCommand{\BaselineClustering}{o m}{\mathcal{C}_{\mathrm{#2}\IfValueT{#1}{,#1}}}
\newcommand{\Similarity}[2]{\rho^{\mathrm{#1}}_{#2}}
